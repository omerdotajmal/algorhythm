// Java — ActivitySelection.java List<Activity> select(Activity[] activities) { // Sort by finish time Arrays.sort(activities, (a, b) -> a.finish - b.finish); List<Activity> selected = new ArrayList<>(); selected.add(activities[0]); // always pick first int lastFinish = activities[0].finish; for (int i = 1; i < activities.length; i++) { if (activities[i].start >= lastFinish) { selected.add(activities[i]); lastFinish = activities[i].finish; } } return selected; }
